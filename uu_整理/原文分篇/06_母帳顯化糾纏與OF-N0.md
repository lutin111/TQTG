"""

Check dt-convergence, Kepler control, and Richardson extrapolation.

import numpy as np
from scipy.integrate import solve_ivp

mu_E = 3.986004418e14
a_orb = 29600e3
ecc   = 0.16
A_T   = 1.0e-10
R0    = 3.0e7

def a_T_mag(r_mag, A):
    if A == 0.0:
        return 0.0
    x = (r_mag / R0)**2
    return A * x / (1.0 + x)

def rhs(t, y, A):
    rx, ry, rz, vx, vy, vz = y
    r_mag = np.sqrt(rx**2 + ry**2 + rz**2)
    a_grav = -mu_E / r_mag**3 * np.array([rx, ry, rz])
    a_rad  = a_T_mag(r_mag, A)
    a_tqtg = a_rad * np.array([rx, ry, rz]) / r_mag
    a = a_grav + a_tqtg
    return [vx, vy, vz, a[0], a[1], a[2]]

def energy(y):
    r = np.array(y[:3])
    v = np.array(y[3:])
    r_mag = np.linalg.norm(r)
    return 0.5*np.dot(v,v) - mu_E/r_mag

def angular_momentum(y):
    r = np.array(y[:3])
    v = np.array(y[3:])
    return np.cross(r, v)

r_p = a_orb * (1 - ecc)
v_p = np.sqrt(mu_E * (1 + ecc) / (a_orb * (1 - ecc)))
y0  = [r_p, 0, 0, 0, v_p, 0]

T_span = 30 * 86400.0

print("=" * 72)
print("OF-N0: numerical convergence guard")
print("=" * 72)
print()

print("Step 1: dt-convergence (TQTG orbit, A_T = 1e-10)")

print("-" * 72)

for dt in dt_list:

                    t_eval=t_eval, rtol=1e-13, atol=1e-9, args=(A_T,))
