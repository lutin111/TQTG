"""
TQTG radial force projected onto ECOM2 basis (D/Y/B).
Direct algebraic projection, no dynamics refit.
"""
import numpy as np
import matplotlib.pyplot as plt

# ---------- Parameters ----------
mu_E = 3.986004418e14
a_orb = 29600e3
ecc = 0.16
A_T = 1.0e-10
R0 = 3.0e7
T_span = 7 * 86400.0
dt = 300.0

# ---------- Orbit (Kepler) ----------
def make_orbit(a, e, T, dt):
    t = np.arange(0, T, dt)
    n = np.sqrt(mu_E / a**3)
    M = n * t
    E = M.copy()
    for _ in range(100):
        dE = (E - e*np.sin(E) - M) / (1 - e*np.cos(E))
        E -= dE
        if np.max(np.abs(dE)) < 1e-14:
            break
    x = a * (np.cos(E) - e)
    y = a * np.sqrt(1 - e**2) * np.sin(E)
    vx = -a*n*np.sin(E) / (1 - e*np.cos(E))
    vy = a*n*np.sqrt(1-e**2)*np.cos(E) / (1 - e*np.cos(E))
    return t, np.column_stack([x, y, np.zeros_like(x)]), \
           np.column_stack([vx, vy, np.zeros_like(vx)]), n

t, r_vec, v_vec, n_mean = make_orbit(a_orb, ecc, T_span, dt)
r_mag = np.linalg.norm(r_vec, axis=1)
r_hat = r_vec / r_mag[:, None]

# ---------- TQTG radial acceleration ----------
def a_T(r_mag, A, R0):
    x = (r_mag / R0)**2
    return A * x / (1 + x)

a_TQTG_mag = a_T(r_mag, A_T, R0)
a_TQTG_vec = a_TQTG_mag[:, None] * r_hat

# ---------- Sun direction (fixed approximation for 7 days) ----------
sun_dir = np.array([0.8, 0.5, 0.3])
sun_dir = sun_dir / np.linalg.norm(sun_dir)
D_hat = np.tile(sun_dir, (len(t), 1))

# ---------- Nominal yaw-steering attitude ----------
Z_hat = -r_hat
Y_hat = np.cross(Z_hat, D_hat)
Y_hat = Y_hat / np.linalg.norm(Y_hat, axis=1)[:, None]
B_hat = np.cross(D_hat, Y_hat)

# ---------- Project TQTG force onto D/Y/B ----------
a_D = np.sum(a_TQTG_vec * D_hat, axis=1)
a_Y = np.sum(a_TQTG_vec * Y_hat, axis=1)
a_B = np.sum(a_TQTG_vec * B_hat, axis=1)

# ---------- ECOM2 fit ----------
def ecom2_fit(t, y, n, which):
    if which == "D":
        cols = [np.ones_like(t), np.cos(2*n*t), np.sin(2*n*t)]
        labels = ["D0", "D2c", "D2s"]
    elif which == "Y":
        cols = [np.ones_like(t)]
        labels = ["Y0"]
    elif which == "B":
        cols = [np.ones_like(t), np.cos(n*t), np.sin(n*t)]
        labels = ["B0", "B1c", "B1s"]
    X = np.column_stack(cols)
    c, *_ = np.linalg.lstsq(X, y, rcond=None)
    y_fit = X @ c
    y_res = y - y_fit
    return c, y_fit, y_res, labels

c_D, fit_D, res_D, lab_D = ecom2_fit(t, a_D, n_mean, "D")
c_Y, fit_Y, res_Y, lab_Y = ecom2_fit(t, a_Y, n_mean, "Y")
c_B, fit_B, res_B, lab_B = ecom2_fit(t, a_B, n_mean, "B")

# ---------- Residual magnitude ----------
res_vec = (res_D[:, None] * D_hat +
           res_Y[:, None] * Y_hat +
           res_B[:, None] * B_hat)
res_mag = np.linalg.norm(res_vec, axis=1)
a_TQTG_norm = np.linalg.norm(a_TQTG_vec, axis=1)

# ---------- Report ----------
print("=" * 60)
print("TQTG force projection onto ECOM2 basis")
print("=" * 60)
print(f"mean |a_TQTG|      = {np.mean(a_TQTG_norm):.4e} m/s^2")
print(f"max  |a_TQTG|      = {np.max(a_TQTG_norm):.4e} m/s^2")
print()
print(f"mean residual      = {np.mean(res_mag):.4e} m/s^2")
print(f"max  residual      = {np.max(res_mag):.4e} m/s^2")
print(f"relative residual  = {np.mean(res_mag)/np.mean(a_TQTG_norm):.4f}")
print()

print("ECOM2 coefficients (D component):")
for lab, c in zip(lab_D, c_D):
    print(f"  {lab:>4s} = {c:+.4e}")
print()
print("ECOM2 coefficients (Y component):")
for lab, c in zip(lab_Y, c_Y):
    print(f"  {lab:>4s} = {c:+.4e}")
print()
print("ECOM2 coefficients (B component):")
for lab, c in zip(lab_B, c_B):
    print(f"  {lab:>4s} = {c:+.4e}")
print()

# ---------- Decomposition of residual into R/T/N ----------
h_vec = np.cross(r_vec, v_vec)
h_hat = h_vec / np.linalg.norm(h_vec, axis=1)[:, None]
t_hat = np.cross(h_hat, r_hat)

res_R = np.sum(res_vec * r_hat, axis=1)
res_T = np.sum(res_vec * t_hat, axis=1)
res_N = np.sum(res_vec * h_hat, axis=1)

print("Residual decomposed into orbital R/T/N:")
print(f"  RMS R = {np.sqrt(np.mean(res_R**2)):.4e} m/s^2")
print(f"  RMS T = {np.sqrt(np.mean(res_T**2)):.4e} m/s^2")
print(f"  RMS N = {np.sqrt(np.mean(res_N**2)):.4e} m/s^2")
print()

# ---------- Plot ----------
fig, axes = plt.subplots(3, 1, figsize=(10, 9), sharex=True)
axes[0].plot(t/86400, a_D*1e11, lw=0.8, label="raw a_D")
axes[0].plot(t/86400, fit_D*1e11, 'r--', lw=0.8, label="ECOM2 fit")
axes[0].plot(t/86400, res_D*1e11, 'k-', lw=0.5, label="residual")
axes[0].set_ylabel(r"$a_D$ [$10^{-11}$ m/s$^2$]")
axes[0].legend(); axes[0].grid(alpha=0.3)

axes[1].plot(t/86400, a_B*1e11, lw=0.8, label="raw a_B")
axes[1].plot(t/86400, fit_B*1e11, 'r--', lw=0.8, label="ECOM2 fit")
axes[1].plot(t/86400, res_B*1e11, 'k-', lw=0.5, label="residual")
axes[1].set_ylabel(r"$a_B$ [$10^{-11}$ m/s$^2$]")
axes[1].legend(); axes[1].grid(alpha=0.3)

axes[2].plot(t/86400, res_R*1e11, label="res R")
axes[2].plot(t/86400, res_T*1e11, label="res T")
axes[2].plot(t/86400, res_N*1e11, label="res N")
axes[2].set_xlabel("t [day]")
axes[2].set_ylabel("residual [orbital R/T/N]")

plt.tight_layout()
plt.savefig("tqtg_ECOM2_projection.png", dpi=140)
