"""
OF-P3a.1: phase-start scan
Same TQTG truth, same pure-GR 6-param refit.
Vary arc start phase u0 for arc lengths 7, 14, 30 days.
"""

import numpy as np
from scipy.integrate import solve_ivp
from scipy.optimize import least_squares

mu_E = 3.986004418e14
a_orb = 29600e3
ecc   = 0.16
A_T   = 1.0e-10
R0    = 3.0e7
n_mean = np.sqrt(mu_E / a_orb**3)
T_orbital = 2*np.pi / n_mean

def a_T_mag(r_mag, A):
    if A == 0.0:
        return 0.0
    x = (r_mag / R0)**2
    return A * x / (1.0 + x)

def rhs(t, y, A):
    rx, ry, rz, vx, vy, vz = y
    r_mag = np.sqrt(rx**2 + ry**2 + rz**2)
    a_grav = -mu_E / r_mag**3 * np.array([rx, ry, rz])
    a_rad  = a_T_mag(r_mag, A)
    a_tqtg = a_rad * np.array([rx, ry, rz]) / r_mag
    a = a_grav + a_tqtg
    return [vx, vy, vz, a[0], a[1], a[2]]

# ---------- Generate long TQTG truth orbit ----------
T_long = 90 * 86400.0
t_dense = np.linspace(0, T_long, 90000)

r_p = a_orb * (1 - ecc)
v_p = np.sqrt(mu_E * (1 + ecc) / (a_orb * (1 - ecc)))
y0  = [r_p, 0, 0, 0, v_p, 0]

sol = solve_ivp(rhs, (0, T_long), y0, method="DOP853",
                t_eval=t_dense, rtol=1e-12, atol=1e-8, args=(A_T,))
r_truth = sol.y[:3, :].T
v_truth = sol.y[3:, :].T

# ---------- Sub-arc refit ----------
def refit_subarc(t_start, T_arc, n_points=20000):
    t_sub = np.linspace(t_start, t_start + T_arc, n_points)
    idx = np.searchsorted(t_dense, t_sub)
    idx = np.clip(idx, 0, len(t_dense)-1)
    r_sub = r_truth[idx]
    v_sub = v_truth[idx]

    y0_fit0 = np.concatenate([r_sub[0], v_sub[0]])

    def gr_traj(y0_fit):
        sol = solve_ivp(rhs, (t_start, t_start + T_arc), y0_fit,
                        method="DOP853", t_eval=t_sub,
                        rtol=1e-12, atol=1e-8, args=(0.0,))
        return sol.y[:3, :].T

    def residuals(y0_fit):
        return (gr_traj(y0_fit) - r_sub).ravel()

    result = least_squares(residuals, y0_fit0, method="lm",
                           xtol=1e-14, ftol=1e-14, gtol=1e-14,
                           max_nfev=200)
    r_fit = gr_traj(result.x)
    dr = r_fit - r_sub
    rms = np.sqrt(np.mean(np.sum(dr**2, axis=1)))
    return rms

# ---------- Scan ----------
arc_lengths = [7, 14, 30]          # days
n_phases = 8
phases = np.linspace(0, 2*np.pi, n_phases+1)[:-1]

print("=" * 78)
print("OF-P3a.1: phase-start scan")
print("=" * 78)

header = f"{'T_arc [d]':>10}"
for u0 in phases:
    header += f"{'u0='+f'{u0/np.pi:.2f}π':>12}"
header += f"{'median':>12}{'max':>12}"
print(header)
print("-" * 78)

for T_d in arc_lengths:
    T_arc = T_d * 86400.0
    row = []
    for u0 in phases:
        t_start = u0 / n_mean
        if t_start + T_arc > T_long:
            row.append(np.nan)
            continue
        rms = refit_subarc(t_start, T_arc)
        row.append(rms)
    valid = [r for r in row if not np.isnan(r)]
    line = f"{T_d:>10d}"
    for rms in row:
        line += f"{rms*1e3:>12.3f}" if not np.isnan(rms) else f"{'---':>12}"
    line += f"{np.median(valid)*1e3:>12.3f}{np.max(valid)*1e3:>12.3f}"
    print(line)

print()
print("All values in mm (post-fit RMS position residual)")
